const API_BASE = window.EPIC_ANALYST_API_BASE || ''; const chatEl = document.getElementById('chat'); const form = document.getElementById('composer'); const questionEl = document.getElementById('question'); const sendBtn = document.getElementById('send'); const clientSelect = document.getElementById('clientSelect'); const suggestionsEl = document.getElementById('suggestions'); function addMessage(role, text, { isError = false } = {}) { const msg = document.createElement('div'); msg.className = `msg ${role}`; const bubble = document.createElement('div'); bubble.className = 'bubble' + (isError ? ' error' : ''); bubble.textContent = text; msg.appendChild(bubble); chatEl.appendChild(msg); chatEl.scrollTop = chatEl.scrollHeight; return bubble; } function addTyping() { const msg = document.createElement('div'); msg.className = 'msg bot'; msg.id = 'typing-indicator'; const bubble = document.createElement('div'); bubble.className = 'bubble'; bubble.innerHTML = '
'; msg.appendChild(bubble); chatEl.appendChild(msg); chatEl.scrollTop = chatEl.scrollHeight; } function removeTyping() { const el = document.getElementById('typing-indicator'); if (el) el.remove(); } function addMeta(bubble, metrics) { if (!metrics) return; const meta = document.createElement('div'); meta.className = 'meta'; meta.innerHTML = ` ⏱ ${metrics.latency_ms} ms 🔤 ${metrics.input_tokens + metrics.output_tokens} tokens 🛠 ${metrics.num_mcp_tool_calls} consultas a Metabase 💵 $${metrics.estimated_cost_usd.toFixed(5)} `; bubble.parentElement.appendChild(meta); } async function loadClients() { try { const res = await fetch(`${API_BASE}/api/clients`); const clients = await res.json(); clientSelect.innerHTML = clients .map((c) => ``) .join(''); } catch (err) { clientSelect.innerHTML = ''; } } async function sendQuestion(question) { if (!question.trim()) return; addMessage('user', question); questionEl.value = ''; sendBtn.disabled = true; addTyping(); try { const res = await fetch(`${API_BASE}/api/chat`, { method: 'POST', headers: { 'Content-Type': 'application/json' }, body: JSON.stringify({ question, client_id: clientSelect.value || 'default', }), }); const data = await res.json(); removeTyping(); if (!res.ok) { addMessage('bot', data.error || 'Ocurrió un error.', { isError: true }); return; } const bubble = addMessage('bot', data.answer); addMeta(bubble, data.metrics); } catch (err) { removeTyping(); addMessage('bot', 'No se pudo contactar al backend. ¿Está corriendo en ' + API_BASE + '?', { isError: true }); } finally { sendBtn.disabled = false; } } form.addEventListener('submit', (e) => { e.preventDefault(); sendQuestion(questionEl.value); }); questionEl.addEventListener('keydown', (e) => { if (e.key === 'Enter' && !e.shiftKey) { e.preventDefault(); form.requestSubmit(); } }); suggestionsEl.addEventListener('click', (e) => { const btn = e.target.closest('button[data-q]'); if (btn) sendQuestion(btn.dataset.q); }); loadClients();
